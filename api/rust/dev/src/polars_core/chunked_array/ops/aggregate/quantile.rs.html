21/// Returns the 0-based indices into `n_nonnull` sorted non-null values that `quantile` needs, and
22/// the fraction of the way from the lower to the upper one: `(lower_idx, upper_idx, fraction)`.
23///
24/// `upper_idx == lower_idx` unless `method` combines two distinct values. Only `Linear` uses
25/// `fraction`; `Midpoint` averages the two values.
26pub fn quantile_indices(
27    n_nonnull: usize,
28    quantile: f64,
29    method: QuantileMethod,
30) -> (usize, usize, f64) {
31    assert!(n_nonnull > 0);
32    let float_idx = (n_nonnull as f64 - 1.0) * quantile;
33    let floor_idx = float_idx as usize;
34    let ceil_idx = (float_idx.ceil() as usize).min(n_nonnull - 1);
35    match method {
36        QuantileMethod::Nearest => {
37            let idx = float_idx.round() as usize;
38            (idx, idx, 0.0)
39        },
40        QuantileMethod::Lower => (floor_idx, floor_idx, 0.0),
41        QuantileMethod::Higher => (ceil_idx, ceil_idx, 0.0),
42        QuantileMethod::Midpoint | QuantileMethod::Linear => {
43            (floor_idx, ceil_idx, float_idx - floor_idx as f64)
44        },
45        QuantileMethod::Equiprobable => {
46            let idx = ((n_nonnull as f64 * quantile).ceil() - 1.0).max(0.0) as usize;
47            (idx, idx, 0.0)
48        },
49    }
50}

52/// Extracts the quantiles of the `quantile` expression input, a single quantile or a list of them.
53pub fn extract_quantiles(quantile: &Series) -> PolarsResult<Vec<f64>> {
54    polars_ensure!(!quantile.is_empty(), ComputeError:
55        "the 'quantile' expression input should produce a single quantile or a list of quantiles, \
56        got an empty input"
57    );
58    polars_ensure!(quantile.len() == 1, ComputeError:
59        "polars does not support varying quantiles yet, \
60        make sure the 'quantile' expression input produces a single quantile or a list of quantiles"
61    );
62    let quantiles = match quantile.dtype() {
63        DataType::List(_) => {
64            let inner = quantile.list()?.get_as_series(0).ok_or_else(
65                || polars_err!(ComputeError: "quantile expression contains null values"),
66            )?;
67            polars_ensure!(!inner.has_nulls(), ComputeError: "quantile expression contains null values");
68            inner
69                .cast(&DataType::Float64)?
70                .f64()?
71                .into_no_null_iter()
72                .collect()
73        },
74        _ => vec![quantile.get(0)?.try_extract()?],
75    };
76    check_quantiles(&quantiles)?;
77    Ok(quantiles)
78}

80fn check_quantiles(quantiles: &[f64]) -> PolarsResult<()> {
81    for &q in quantiles {
82        polars_ensure!(
83            (0.0..=1.0).contains(&q),
84            ComputeError: "`quantile` should be between 0.0 and 1.0, got {}", q,
85        );
86    }
87    Ok(())
88}

90fn linear_interpol<T: Float>(lower: T, upper: T, fraction: f64) -> T {
91    if lower == upper {
92        lower
93    } else {
94        let proportion: T = T::from(fraction).unwrap();
95        proportion * (upper - lower) + lower
96    }
97}
98fn midpoint_interpol<T: Float>(lower: T, upper: T) -> T {
99    if lower == upper {
100        lower
101    } else {
102        (lower + upper) / (T::one() + T::one())
103    }
104}

106/// Combines the values at the ranks returned by [`quantile_indices`] into the quantile.
107fn interpolate<T: ToPrimitive>(
108    lower: T,
109    upper: T,
110    fraction: f64,
111    method: QuantileMethod,
112) -> Option<f64> {
113    let lower = lower.to_f64()?;
114    match method {
115        QuantileMethod::Midpoint => Some(midpoint_interpol(lower, upper.to_f64()?)),
116        QuantileMethod::Linear => Some(linear_interpol(lower, upper.to_f64()?, fraction)),
117        _ => Some(lower),
118    }
119}

121/// Computes `quantile` of `n_nonnull` values, where `order_stats(ranks, out)` writes the values at
122/// the given ascending ranks of the sorted values to `out`.
123fn quantile_with<T: ToPrimitive + Copy + Default>(
124    n_nonnull: usize,
125    quantile: f64,
126    method: QuantileMethod,
127    order_stats: impl FnOnce(&[usize], &mut [T]),
128) -> Option<f64> {
129    if n_nonnull == 0 {
130        return None;
131    }
132    let (lower, upper, fraction) = quantile_indices(n_nonnull, quantile, method);
133    let k = if lower == upper { 1 } else { 2 };
134    let mut vals = [T::default(); 2];
135    order_stats(&[lower, upper][..k], &mut vals[..k]);
136    interpolate(vals[0], vals[k - 1], fraction, method)
137}

139/// [`quantile_with`] for several quantiles, fetching all the order statistics they need at once.
140fn quantiles_with<T: ToPrimitive + Copy + Default>(
141    n_nonnull: usize,
142    quantiles: &[f64],
143    method: QuantileMethod,
144    order_stats: impl FnOnce(&[usize], &mut [T]),
145) -> Vec<Option<f64>> {
146    if n_nonnull == 0 {
147        return vec![None; quantiles.len()];
148    }
149    let indices: Vec<_> = quantiles
150        .iter()
151        .map(|&q| quantile_indices(n_nonnull, q, method))
152        .collect();
153    let mut ranks: Vec<usize> = indices.iter().flat_map(|&(l, u, _)| [l, u]).collect();
154    ranks.sort_unstable();
155    ranks.dedup();
156    let mut vals = vec![T::default(); ranks.len()];
157    order_stats(&ranks, &mut vals);
158    let at = |r| vals[ranks.binary_search(&r).unwrap()];
159    indices
160        .iter()
161        .map(|&(lower, upper, fraction)| interpolate(at(lower), at(upper), fraction, method))
162        .collect()
163}

165/// The `quantile` of `vals`, non-null values in any order, which it reorders.
166pub(crate) fn quantile_of_nonnull<T: ToPrimitive + TotalOrd + Copy + Default>(
167    vals: &mut [T],
168    quantile: f64,
169    method: QuantileMethod,
170) -> Option<f64> {
171    quantile_with(vals.len(), quantile, method, |ranks, out| {
172        select_many_unstable_by(vals, ranks, out, TotalOrd::tot_cmp)
173    })
174}

176/// Reads the values at the ascending `ranks` of the non-null values of `ca` into `out`, given that
177/// `ca` is flagged sorted (descending if `descending`), which keeps all its nulls at one end.
178fn sorted_order_stats<T: PolarsNumericType>(
179    ca: &ChunkedArray<T>,
180    descending: bool,
181    ranks: &[usize],
182    out: &mut [T::Native],
183) {
184    let n_nonnull = ca.len() - ca.null_count();
185    let start = ca.first_non_null().unwrap_or(0);
186    debug_assert!(ranks.last().is_none_or(|&r| r < n_nonnull));
187
188    // Visit the ranks in ascending row order, so a single pass over the chunks finds them.
189    let mut chunks = ca.downcast_iter();
190    let mut chunk = chunks.next().unwrap();
191    let mut chunk_start = 0;
192    for j in 0..ranks.len() {
193        let (i, idx) = if descending {
194            let i = ranks.len() - 1 - j;
195            (i, start + n_nonnull - 1 - ranks[i])
196        } else {
197            (j, start + ranks[j])
198        };
199        while idx >= chunk_start + chunk.len() {
200            chunk_start += chunk.len();
201            chunk = chunks.next().unwrap();
202        }
203        out[i] = chunk.value(idx - chunk_start);
204    }
205}

207/// The `quantile` of `ca`, which is sorted (descending if `descending`) with its nulls at one end.
208pub(crate) fn sorted_quantile<T: PolarsNumericType>(
209    ca: &ChunkedArray<T>,
210    descending: bool,
211    quantile: f64,
212    method: QuantileMethod,
213) -> Option<f64> {
214    let n_nonnull = ca.len() - ca.null_count();
215    quantile_with(n_nonnull, quantile, method, |ranks, out| {
216        sorted_order_stats(ca, descending, ranks, out)
217    })
218}

220/// Writes the values at the strictly ascending `ranks` of the sorted non-null values of `ca` to
221/// `out`, reading them directly if `ca` is flagged sorted and selecting them otherwise.
222pub fn order_statistics<T: PolarsNumericType>(
223    ca: &ChunkedArray<T>,
224    ranks: &[usize],
225    out: &mut [T::Native],
226) {
227    if ranks.is_empty() {
228        return;
229    }
230    match ca.is_sorted_flag() {
231        IsSorted::Not => {
232            select_many_unstable_by(&mut ca.to_vec_non_null(), ranks, out, TotalOrd::tot_cmp)
233        },
234        sorted => sorted_order_stats(ca, sorted == IsSorted::Descending, ranks, out),
235    }
236}

238/// The `quantile` of `ca`, after validating it.
239fn ca_quantile<T: PolarsNumericType>(
240    ca: &ChunkedArray<T>,
241    quantile: f64,
242    method: QuantileMethod,
243) -> PolarsResult<Option<f64>> {
244    check_quantiles(&[quantile])?;
245    let n_nonnull = ca.len() - ca.null_count();
246    Ok(quantile_with(n_nonnull, quantile, method, |ranks, out| {
247        order_statistics(ca, ranks, out)
248    }))
249}

251/// The `quantiles` of `ca`, after validating them.
252fn ca_quantiles<T: PolarsNumericType>(
253    ca: &ChunkedArray<T>,
254    quantiles: &[f64],
255    method: QuantileMethod,
256) -> PolarsResult<Vec<Option<f64>>> {
257    check_quantiles(quantiles)?;
258    let n_nonnull = ca.len() - ca.null_count();
259    Ok(quantiles_with(
260        n_nonnull,
261        quantiles,
262        method,
263        |ranks, out| order_statistics(ca, ranks, out),
264    ))
265}

267/// The `quantiles` of the non-null values of `ca` that are not NaN, after validating them, or
268/// `None` if there are no such values.
269pub fn quantiles_without_nan(
270    ca: &Float64Chunked,
271    quantiles: &[f64],
272    method: QuantileMethod,
273) -> PolarsResult<Option<Vec<f64>>> {
274    check_quantiles(quantiles)?;
275    let n_nan: usize = ca
276        .downcast_iter()
277        .map(|arr| arr.non_null_values_iter().filter(|v| v.is_nan()).count())
278        .sum();
279    // NaN orders above every other value, so the lowest ranks are exactly the values without NaN.
280    let n = ca.len() - ca.null_count() - n_nan;
281    if n == 0 {
282        return Ok(None);
283    }
284    let qs = quantiles_with(n, quantiles, method, |ranks, out| {
285        order_statistics(ca, ranks, out)
286    });
287    Ok(Some(qs.into_iter().map(Option::unwrap).collect()))
288}

290impl<T> ChunkQuantile<f64> for ChunkedArray<T>
291where
292    T: PolarsIntegerType,
293{
294    fn quantile(&self, quantile: f64, method: QuantileMethod) -> PolarsResult<Option<f64>> {
295        ca_quantile(self, quantile, method)
296    }
297
298    fn quantiles(
299        &self,
300        quantiles: &[f64],
301        method: QuantileMethod,
302    ) -> PolarsResult<Vec<Option<f64>>> {
303        ca_quantiles(self, quantiles, method)
304    }
305
306    fn median(&self) -> Option<f64> {
307        self.quantile(0.5, QuantileMethod::Linear).unwrap() // unwrap fine since quantile in range
308    }
309}

311macro_rules! impl_chunk_quantile_for_float_chunked {
312    ($T:ty, $CA:ty) => {
313        impl ChunkQuantile<$T> for $CA {
314            fn quantile(&self, quantile: f64, method: QuantileMethod) -> PolarsResult<Option<$T>> {
315                Ok(ca_quantile(self, quantile, method)?.map(AsPrimitive::as_))
316            }
317
318            fn quantiles(
319                &self,
320                quantiles: &[f64],
321                method: QuantileMethod,
322            ) -> PolarsResult<Vec<Option<$T>>> {
323                Ok(ca_quantiles(self, quantiles, method)?
324                    .into_iter()
325                    .map(|opt| opt.map(AsPrimitive::as_))
326                    .collect())
327            }
328
329            fn median(&self) -> Option<$T> {
330                self.quantile(0.5, QuantileMethod::Linear).unwrap() // unwrap fine since quantile in range
331            }
332        }
333    };
334}
